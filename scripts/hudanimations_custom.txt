//Health Bonus Pulse
event HudHealthBonusPulse
{
	Animate PlayerCharaImage				alpha		"0"					Linear 0.0 0.00
	Animate PlayerCharaImageBonus			alpha		"255"				Linear 0.0 0.00

	Animate	PlayerStatusHealthBonusImage 	alpha		"255"				Linear 0.0 0.2
	Animate	PlayerStatusHealthBonusImage 	alpha		"0"					Linear 0.2 0.4

	Animate PlayerStatusHealthValue			fgcolor		"HudBonus" 			Linear 0.0 0.275
	Animate PlayerStatusHealthValue			fgcolor		"HudPrimaryLight" 	Linear 0.225 0.175
	RunEvent HudHealthBonusPulseLoop		0.4
}

// call to loop HudHealthBonusPulse
event HudHealthBonusPulseLoop
{
	RunEvent HudHealthBonusPulse 			0.0
}

event HudHealthBonusPulseStop
{
	StopEvent HudHealthBonusPulse 			0.0
	StopEvent HudHealthBonusPulseLoop 		0.0
	Animate PlayerCharaImageBonus			alpha		"0"					Linear 0.0 0.00
	Animate PlayerCharaImage				alpha		"255"				Linear 0.0 0.00
	Animate PlayerStatusHealthValue			fgcolor		"HudPrimaryLight" 	Linear 0.0 0.275
}
